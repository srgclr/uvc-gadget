# Raspberry Pi 4 firmware configuration for UVC gadget mode
# This file is copied to the FAT boot partition.

# Use 64-bit kernel
arm_64bit=1
kernel=Image

# Enable the dwc2 OTG controller on the USB-C port in peripheral (gadget) mode.
# Without this overlay the USB-C port is power-only.
dtoverlay=dwc2,dr_mode=peripheral

# Use cut-down firmware: no GPU codecs or 3D, smaller file, faster startup
gpu_mem=16
start_file=start4cd.elf
fixup_file=fixup4cd.dat

# Disable bluetooth: no BT firmware is installed, so the hardware
# won't initialise. Suppress the kernel's hci_uart probe attempt.
dtoverlay=disable-bt

# Disable HDMI output: headless appliance, serial console used for debug
hdmi_ignore_hotplug=1

# Disable onboard WiFi chip to skip SDIO probe on mmc1
dtoverlay=disable-wifi

# Firmware startup optimisations
disable_splash=1        # Skip rainbow splash screen
boot_delay=0            # No artificial delay before loading kernel
force_eeprom_read=0     # Skip HAT EEPROM probe on GPIO 0/1
camera_auto_detect=0    # Don't probe for camera modules (CSI)
display_auto_detect=0   # Don't probe for display peripherals
hdmi_ignore_cec_init=1  # Skip CEC bus initialisation
initial_turbo=30        # Hold CPU at max frequency for first 30s of boot
